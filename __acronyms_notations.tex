\section{Acronyms}

\begin{itemize}
    \item \textbf{MFA}: Multi-factor Authentication
    \item \textbf{PII}: Personally Identifiable Information
    \item \textbf{OSI}: Open Systems Interconnection (Context: Networking)
    \item \textbf{CTA}: Credential Tweaking Attack
    \item \textbf{CSA}: Credential Stuffing Attack
    \item \textbf{ICO:} Information commissioner's office (Regulatory body in the UK)
    \item \textbf{CSP:} Content Security Policy
    \item \textbf{DLP:} Discrete Logarithm Problem
    \item \textbf{PQC:} Post-quantum Cryptography
\end{itemize}

\section{Symbols and Notations}

\begin{itemize}
    \item $h(x) = y$ Hash function, where $x$ is the input (message), and $y$ is the output, hash value or digest.
    \item $g() = x$ A generator function that generates a human-chosen and memorable password. 
\end{itemize}
